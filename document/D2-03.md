- `calculatePenalty(uint256 depositId) external view returns (uint256)`.
- NFT Settlement Policy: Terminal certificate **retained** as historical certificate (không burn), re-withdrawal bị chặn bằng `DepositStatus.Withdrawn`.

## Part B — Implementation Summary

### 1. Financial Calculation Formulas

#### 1.1 Full-Term Interest
```text
interest = Math.mulDiv(
    deposit.principal,
    uint256(deposit.aprBpsAtOpen) * tenorSeconds,
    365 days * 10000
)
```
- `tenorSeconds = deposit.maturityAt - deposit.startAt`.
- Lãi đơn (simple interest) cố định theo kỳ gửi ban đầu; rút trễ không sinh thêm lãi.
- Sử dụng snapshot `aprBpsAtOpen`; không lấy APR hiện tại của plan.
- Làm tròn xuống theo floor integer division bằng `Math.mulDiv` (512-bit intermediate multiplication chống overflow).
- Khi APR = 0, lãi trả về đúng 0n.

#### 1.2 Early Withdrawal Penalty
```text
penalty = Math.mulDiv(
    deposit.principal,
    uint256(deposit.penaltyBpsAtOpen),
    10000
)
```
- Sử dụng snapshot `penaltyBpsAtOpen`.
- Làm tròn xuống (floor integer division).
- Không vượt quá principal; khi penalty = 0 bps, penalty = 0; khi penalty = 10000 bps, penalty = principal.
- Early withdrawal tuyệt đối không chi trả bất kỳ khoản lãi nào.

### 2. Maturity Withdrawal Flow (`withdrawAtMaturity`)

1. **Preconditions & Guards:**
   - `nonReentrant` bảo vệ toàn bộ entrypoint.
   - `vaultManager.savingCore() == address(this)` (nếu không, revert `CoreNotBound`).
   - `!vaultManager.paused()` (nếu paused, revert `EnforcedPause`).
   - `1 <= depositId <= depositCount` (nếu không, revert `DepositNotFound`).
   - `deposit.status == DepositStatus.Active` (nếu không, revert `DepositNotActive`).
   - `ownerOf(depositId) == msg.sender` (nếu không, revert `NotDepositOwner`).
   - `block.timestamp >= deposit.maturityAt` (nếu chưa đủ hạn, revert `DepositNotMatured`).
2. **Checks-Effects-Interactions:**
   - Tính toán contractual interest `I = _calculateInterest(deposit)`.
   - Chuyển `deposit.status = DepositStatus.Withdrawn`.
   - Giảm liability: `totalPrincipal -= deposit.principal`.
   - Nếu `I > 0`: gọi `vaultManager.payInterest(I)` (pull `I` MockUSDC từ Vault vào Core).
   - Nếu `I == 0`: bỏ qua gọi `payInterest(0)` để tránh revert `AmountZero` từ Vault.
   - Chuyển `P + I` MockUSDC từ Core đến current NFT owner (`msg.sender`).
   - Emit `Withdrawn(depositId, msg.sender, principal, interest, false)`.

### 3. Early Withdrawal Flow (`earlyWithdraw`)

1. **Preconditions & Guards:**
   - `nonReentrant`, `bound`, `whenNotPaused`.
   - `depositId` hợp lệ, status `Active`, caller là current `ownerOf(depositId)`.
   - `block.timestamp < deposit.maturityAt` (nếu `block.timestamp >= deposit.maturityAt`, revert `EarlyWithdrawalClosed`).
2. **Checks-Effects-Interactions:**
   - Tính penalty `F = _calculatePenalty(deposit)`.
   - Tính net principal payout: `netPayout = deposit.principal - F`.
   - Đọc `feeReceiver = vaultManager.feeReceiver()`.
   - Chuyển `deposit.status = DepositStatus.Withdrawn`.
   - Giảm toàn bộ liability gốc: `totalPrincipal -= deposit.principal`.
   - Nếu `netPayout > 0`: chuyển `netPayout` MockUSDC từ Core đến `msg.sender`.
   - Nếu `F > 0`: chuyển `F` MockUSDC từ Core đến `feeReceiver`.
   - Không gọi `vaultManager.payInterest` (quỹ lãi giữ nguyên 100%).
   - Emit `Withdrawn(depositId, msg.sender, principal, 0, true)`.

### 4. Ownership Authorization & NFT Retention

- Current NFT owner được xác định động qua `ownerOf(depositId)`.
- Khi Alice chuyển NFT cho Bob (`transferFrom`), Bob trở thành người duy nhất có quyền settlement và nhận toàn bộ số tiền thanh toán; Alice hoàn toàn mất quyền.
- Approved operators (dù có ERC721 approval) bị từ chối với `NotDepositOwner` khi gọi rút tiền (thiết kế owner-only chặt chẽ).
- Terminal NFT được **giữ lại** (retained) trong ví của owner làm chứng chỉ lịch sử; các giao dịch rút hoặc tái tục tiếp theo bị chặn bởi status `Withdrawn`.

### Key Code Snippets from `contracts/SavingCore.sol`

```solidity
function withdrawAtMaturity(uint256 depositId) external nonReentrant {
    address actualCore = vaultManager.savingCore();
    if (actualCore != address(this)) revert CoreNotBound(address(this), actualCore);
    if (vaultManager.paused()) revert EnforcedPause();
    if (depositId == 0 || depositId > depositCount) revert DepositNotFound(depositId);

    Deposit storage deposit = _deposits[depositId];
    if (deposit.status != DepositStatus.Active) {
        revert DepositNotActive(depositId, deposit.status);
    }
    if (ownerOf(depositId) != msg.sender) {
        revert NotDepositOwner(depositId, msg.sender);
    }
    if (block.timestamp < deposit.maturityAt) {
        revert DepositNotMatured(depositId, deposit.maturityAt);
    }

    uint256 principal = deposit.principal;
    uint256 interest = _calculateInterest(deposit);

    deposit.status = DepositStatus.Withdrawn;
    totalPrincipal -= principal;

    if (interest > 0) {
        vaultManager.payInterest(interest);
    }

    token.safeTransfer(msg.sender, principal + interest);

    emit Withdrawn(depositId, msg.sender, principal, interest, false);
}

function earlyWithdraw(uint256 depositId) external nonReentrant {
    address actualCore = vaultManager.savingCore();
    if (actualCore != address(this)) revert CoreNotBound(address(this), actualCore);
    if (vaultManager.paused()) revert EnforcedPause();
    if (depositId == 0 || depositId > depositCount) revert DepositNotFound(depositId);

    Deposit storage deposit = _deposits[depositId];
    if (deposit.status != DepositStatus.Active) {
        revert DepositNotActive(depositId, deposit.status);
    }
    if (ownerOf(depositId) != msg.sender) {
        revert NotDepositOwner(depositId, msg.sender);
    }
    if (block.timestamp >= deposit.maturityAt) {
        revert EarlyWithdrawalClosed(depositId, deposit.maturityAt);
    }

    uint256 principal = deposit.principal;
    uint256 penalty = _calculatePenalty(deposit);
    uint256 netPayout = principal - penalty;
    address feeReceiver = vaultManager.feeReceiver();

    deposit.status = DepositStatus.Withdrawn;
    totalPrincipal -= principal;

    if (netPayout > 0) {
        token.safeTransfer(msg.sender, netPayout);
    }

    if (penalty > 0) {
        token.safeTransfer(feeReceiver, penalty);
    }

    emit Withdrawn(depositId, msg.sender, principal, 0, true);
}
```
